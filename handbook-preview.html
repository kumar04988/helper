Server vs Client Component

💡
All components are Server Components by default in Next.js App Router. Add 'use client' only when you need useState, useEffect, or event handlers.
// ✅ Server Component — runs on server, no JS sent to browser
// src/app/products/page.tsx
async function getProducts() {
  const res = await fetch('http://localhost:4000/products', {
    next: { revalidate: 60 }, // Cache and ISR refresh every 60s
  })
  return res.json()
}

export default async function ProductsPage() {
  const products = await getProducts()
  return (
    <div className="grid grid-cols-3 gap-4 p-8">
      {products.map((p) => (
        <ProductCard key={p.id} product={p} />
      ))}
    </div>
  )
}
// ✅ Client Component — handles user interactions
// src/components/product-card.tsx
'use client'

import { useState } from 'react'

export function ProductCard({ product }: { product: { id: string; name: string; price: number } }) {
  const [added, setAdded] = useState(false)

  const addToCart = async () => {
    await fetch('/api/cart', {
      method: 'POST',
      headers: { 'Content-Type': 'application/json' },
      body: JSON.stringify({ productId: product.id }),
    })
    setAdded(true)
  }

  return (
    <div className="border rounded-xl p-4 shadow-sm">
      <h3 className="font-semibold">{product.name}</h3>
      <p className="text-slate-600">${product.price}</p>
      <button
        onClick={addToCart}
        disabled={added}
        className="mt-3 w-full bg-blue-600 text-white py-2 rounded-lg disabled:opacity-50"
      >
        {added ? 'Added ✓' : 'Add to Cart'}
      </button>
    </div>
  )
}

React Query — fetch data the right way

// src/hooks/use-todos.ts
import { useQuery, useMutation, useQueryClient } from '@tanstack/react-query'
import { useAppStore } from '@/lib/store'

const API = process.env.NEXT_PUBLIC_API_URL

export function useTodos() {
  const accessToken = useAppStore((s) => s.accessToken)

  return useQuery({
    queryKey: ['todos'],
    queryFn: async () => {
      const res = await fetch(`${API}/todos`, {
        headers: { Authorization: `Bearer ${accessToken}` },
      })
      if (!res.ok) throw new Error('Failed to fetch todos')
      return res.json()
    },
    enabled: !!accessToken,
    staleTime: 30_000,
  })
}

export function useCreateTodo() {
  const queryClient = useQueryClient()
  const accessToken = useAppStore((s) => s.accessToken)

  return useMutation({
    mutationFn: async (data: { title: string; priority: string }) => {
      const res = await fetch(`${API}/todos`, {
        method: 'POST',
        headers: {
          'Content-Type': 'application/json',
          Authorization: `Bearer ${accessToken}`,
        },
        body: JSON.stringify(data),
      })
      if (!res.ok) throw new Error('Failed to create todo')
      return res.json()
    },
    onSuccess: () => queryClient.invalidateQueries({ queryKey: ['todos'] }),
  })
}

Forms — React Hook Form + Zod

// src/components/create-todo-form.tsx
'use client'

import { useForm } from 'react-hook-form'
import { zodResolver } from '@hookform/resolvers/zod'
import { z } from 'zod'
import { useCreateTodo } from '@/hooks/use-todos'

const todoSchema = z.object({
  title: z.string().min(1, 'Title is required').max(200),
  priority: z.enum(['low', 'medium', 'high']).default('medium'),
})
type TodoFormValues = z.infer<typeof todoSchema>

export function CreateTodoForm() {
  const { mutate: createTodo, isPending, error } = useCreateTodo()
  const { register, handleSubmit, reset, formState: { errors } } =
    useForm<TodoFormValues>({ resolver: zodResolver(todoSchema) })

  const onSubmit = (data: TodoFormValues) => {
    createTodo(data, { onSuccess: () => reset() })
  }

  return (
    <form onSubmit={handleSubmit(onSubmit)} className="flex gap-2">
      <div className="flex-1">
        <input
          {...register('title')}
          placeholder="What needs to be done?"
          className="w-full border rounded-lg px-3 py-2"
        />
        {errors.title && <p className="text-red-500 text-sm mt-1">{errors.title.message}</p>}
      </div>
      <button type="submit" disabled={isPending}
        className="bg-blue-600 text-white px-4 py-2 rounded-lg disabled:opacity-50">
        {isPending ? 'Adding...' : 'Add'}
      </button>
    </form>
  )
}

Complete Fastify server

// api/src/index.ts
import Fastify from 'fastify'
import cors from '@fastify/cors'
import helmet from '@fastify/helmet'
import { authRoutes } from './routes/auth'
import { todoRoutes } from './routes/todos'
import { prisma } from './lib/prisma'

const server = Fastify({ logger: true })

await server.register(helmet)
await server.register(cors, {
  origin: process.env.ALLOWED_ORIGINS?.split(',') ?? ['http://localhost:3000'],
  credentials: true,
})

// Health check — no auth, always live
server.get('/health', async () => ({
  status: 'ok',
  timestamp: new Date().toISOString(),
  db: await prisma.$queryRaw`SELECT 1`.then(() => 'ok').catch(() => 'error'),
}))

await server.register(authRoutes, { prefix: '/auth' })
await server.register(todoRoutes, { prefix: '/todos' })

// Global error handler
server.setErrorHandler((error, _req, reply) => {
  if (error.name === 'ZodError') {
    return reply.status(400).send({ error: 'Validation failed', code: 'VALIDATION_ERROR' })
  }
  const code = error.statusCode ?? 500
  return reply.status(code).send({
    error: code === 500 ? 'Internal server error' : error.message,
  })
})

process.on('SIGTERM', async () => {
  await server.close()
  await prisma.$disconnect()
  process.exit(0)
})

await server.listen({ port: Number(process.env.PORT) || 4000, host: '0.0.0.0' })

CRUD routes

// api/src/routes/todos.ts
import { FastifyPluginAsync } from 'fastify'
import { z } from 'zod'
import { TodoService } from '../services/todo.service'
import { authenticate } from '../middleware/auth'

const createTodoSchema = z.object({
  title: z.string().min(1).max(200),
  priority: z.enum(['low', 'medium', 'high']).default('medium'),
})

export const todoRoutes: FastifyPluginAsync = async (fastify) => {
  const svc = new TodoService()

  fastify.get('/', { preHandler: [authenticate] }, async (req, reply) => {
    return reply.send(await svc.findAll(req.user.userId))
  })

  fastify.post('/', { preHandler: [authenticate] }, async (req, reply) => {
    const body = createTodoSchema.parse(req.body)
    return reply.status(201).send(await svc.create({ ...body, userId: req.user.userId }))
  })

  fastify.patch('/:id', { preHandler: [authenticate] }, async (req, reply) => {
    const { id } = req.params as { id: string }
    const body = z.object({ completed: z.boolean().optional() }).parse(req.body)
    const todo = await svc.update(id, req.user.userId, body)
    if (!todo) return reply.status(404).send({ error: 'Not found' })
    return reply.send(todo)
  })

  fastify.delete('/:id', { preHandler: [authenticate] }, async (req, reply) => {
    const { id } = req.params as { id: string }
    await svc.delete(id, req.user.userId)
    return reply.status(204).send()
  })
}

Complete auth routes

// api/src/routes/auth.ts
import bcrypt from 'bcrypt'
import { FastifyPluginAsync } from 'fastify'
import { z } from 'zod'
import { prisma } from '../lib/prisma'
import { generateTokens } from '../lib/jwt'
import { authenticate } from '../middleware/auth'

const registerSchema = z.object({
  email: z.string().email(),
  password: z.string().min(8, 'Password must be at least 8 characters'),
  name: z.string().min(1).max(100),
})

export const authRoutes: FastifyPluginAsync = async (fastify) => {
  fastify.post('/register', async (req, reply) => {
    const { email, password, name } = registerSchema.parse(req.body)
    const existing = await prisma.user.findUnique({ where: { email } })
    if (existing) return reply.status(409).send({ error: 'Email already in use' })

    const passwordHash = await bcrypt.hash(password, 12)
    const user = await prisma.user.create({ data: { email, passwordHash, name } })
    const tokens = await generateTokens(user.id)
    return reply.status(201).send({ user: { id: user.id, email, name }, ...tokens })
  })

  fastify.post('/login', async (req, reply) => {
    const { email, password } = z.object({
      email: z.string().email(), password: z.string()
    }).parse(req.body)

    const user = await prisma.user.findUnique({ where: { email } })
    // Constant-time comparison — prevents email enumeration
    const valid = user
      ? await bcrypt.compare(password, user.passwordHash)
      : await bcrypt.compare(password, '$2b$12$invalidhashfortimingattack')

    if (!user || !valid) {
      return reply.status(401).send({ error: 'Invalid email or password' })
    }

    const tokens = await generateTokens(user.id)
    return reply.send({ user: { id: user.id, email, name: user.name }, ...tokens })
  })

  fastify.post('/refresh', async (req, reply) => {
    const { refreshToken } = req.body as { refreshToken: string }
    const session = await prisma.session.findUnique({
      where: { refreshToken }, include: { user: true }
    })
    if (!session || session.expiresAt < new Date()) {
      if (session) await prisma.session.delete({ where: { id: session.id } })
      return reply.status(401).send({ error: 'Invalid or expired refresh token' })
    }
    const { accessToken } = await generateTokens(session.userId)
    return reply.send({ accessToken })
  })

  fastify.get('/me', { preHandler: [authenticate] }, async (req, reply) => {
    const user = await prisma.user.findUnique({
      where: { id: req.user.userId },
      select: { id: true, email: true, name: true, role: true },
    })
    if (!user) return reply.status(404).send({ error: 'Not found' })
    return reply.send(user)
  })
}

Auth middleware

// api/src/middleware/auth.ts
import jwt from 'jsonwebtoken'
import { FastifyRequest, FastifyReply } from 'fastify'

declare module 'fastify' {
  interface FastifyRequest { user: { userId: string } }
}

export async function authenticate(req: FastifyRequest, reply: FastifyReply) {
  const token = req.headers.authorization?.replace('Bearer ', '')
  if (!token) return reply.status(401).send({ error: 'Token required' })

  try {
    const payload = jwt.verify(token, process.env.JWT_SECRET!) as { userId: string }
    req.user = { userId: payload.userId }
  } catch (err) {
    if (err instanceof jwt.TokenExpiredError)
      return reply.status(401).send({ error: 'Token expired', code: 'TOKEN_EXPIRED' })
    return reply.status(401).send({ error: 'Invalid token' })
  }
}

  it('GET /todos returns empty array', async () => {

      method: 'GET', url: '/todos',
      headers: { Authorization: `Bearer ${token}` },
    })
    expect(res.statusCode).toBe(200)
    expect(JSON.parse(res.body)).toEqual([])
  })

  it('POST /todos creates todo', async () => {
    const res = await app.inject({
      method: 'POST', url: '/todos',
      headers: { Authorization: `Bearer ${token}`, 'Content-Type': 'application/json' },
      body: JSON.stringify({ title: 'Test todo', priority: 'high' }),
    })
    expect(res.statusCode).toBe(201)
    expect(JSON.parse(res.body).title).toBe('Test todo')
  })

  it('POST /todos returns 401 without token', async () => {
    const res = await app.inject({ method: 'POST', url: '/todos' })
    expect(res.statusCode).toBe(401)
  })

Production Launch Checklist
Run this before every deploy. Ship only when all critical boxes are green.
✅
Check every box before going live. Boxes marked critical are blocking — do not ship with these unresolved.

Frontend (critical)

  • Loading, error, and empty states on every data view
  • Env vars set in hosting platform (not hardcoded)
  • No API keys or secrets in client bundle
  • Forms disabled during submit, validation errors shown

Backend (critical)

  • Auth enforced on all private routes
  • Input validation on all write endpoints
  • GET /health returns 200
  • Global error handler — no stack traces to clients
  • Graceful shutdown on SIGTERM

Database (critical)

  • Migrations applied successfully
  • Automated backups enabled
  • Indexes on columns used in WHERE and JOIN

Security (critical)

  • HTTPS with valid certificate
  • Secrets in env/vault, not in code or git
  • Rate limiting on auth endpoints

Observability (critical)

  • Logs visible in aggregator
  • Alert on 5xx spike and health failure

Deploy safety (critical)

  • Rollback procedure documented (previous image tag)
  • Staging smoke test: health + login + create item
  • DB migrations ran BEFORE app deployed
